cmake_minimum_required(VERSION 3.16)
project(lc3_isa3c VERSION 1.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(MINGW)
  add_link_options(-static)
endif()
add_library(image_io src/image.c)
target_include_directories(image_io PUBLIC include)
add_library(assembler_lib src/assembler.c src/linker.c)
target_link_libraries(assembler_lib PUBLIC image_io)
add_executable(ensamblador src/assembler_main.c)
target_link_libraries(ensamblador PRIVATE assembler_lib)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/os_image.h ${CMAKE_CURRENT_BINARY_DIR}/sistema.obj
  COMMAND ensamblador ${CMAKE_CURRENT_SOURCE_DIR}/ejemplos/sistema.asm
          -o ${CMAKE_CURRENT_BINARY_DIR}/sistema.obj --emit-c ${CMAKE_CURRENT_BINARY_DIR}/os_image.h
  DEPENDS ensamblador ejemplos/sistema.asm VERBATIM)
add_library(cpu_core src/lc3.c src/os.c ${CMAKE_CURRENT_BINARY_DIR}/os_image.h)
target_include_directories(cpu_core PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(cpu_core PUBLIC image_io)
add_library(console_io src/console.c)
target_include_directories(console_io PUBLIC src)
add_executable(simulador src/simulator_main.c)
target_link_libraries(simulador PRIVATE cpu_core console_io)
add_executable(lc3 src/main.c)
target_link_libraries(lc3 PRIVATE cpu_core assembler_lib console_io)
add_library(gui_session src/gui_session.c)
target_link_libraries(gui_session PUBLIC cpu_core assembler_lib)
if(WIN32)
  enable_language(RC)
  add_executable(isa3c_gui WIN32 src/gui_win32.c src/gui_editor.c resources/isa3c.rc)
  target_include_directories(isa3c_gui PRIVATE resources)
  target_link_libraries(isa3c_gui PRIVATE gui_session comctl32 comdlg32 shell32 ole32 gdi32 dwmapi uuid)
  if(MINGW)
    target_link_options(isa3c_gui PRIVATE -municode)
  endif()
endif()
file(GLOB BENCH_MODULES CONFIGURE_DEPENDS benchmark/modulos/*.asm)
foreach(PROGRAM funciones calculadora calculadora_automatica)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${PROGRAM}.obj
    COMMAND ensamblador ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/${PROGRAM}.asm
            -I ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/modulos -o ${CMAKE_CURRENT_BINARY_DIR}/${PROGRAM}.obj
    DEPENDS ensamblador benchmark/${PROGRAM}.asm ${BENCH_MODULES} VERBATIM)
  list(APPEND BENCH_OBJECTS ${CMAKE_CURRENT_BINARY_DIR}/${PROGRAM}.obj)
endforeach()
add_custom_target(benchmark_objects ALL DEPENDS ${BENCH_OBJECTS})
include(CTest)
if(BUILD_TESTING)
  if(WIN32)
    add_executable(test_gui_editor tests/test_gui_editor.c src/gui_editor.c)
    target_include_directories(test_gui_editor PRIVATE include)
    target_link_libraries(test_gui_editor PRIVATE comctl32)
    add_test(NAME editor_pegado COMMAND test_gui_editor)
  endif()
  add_executable(test_gui_session tests/test_gui_session.c)
  target_link_libraries(test_gui_session PRIVATE gui_session)
  add_dependencies(test_gui_session benchmark_objects)
  add_test(NAME interfaz COMMAND test_gui_session ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/gui_prueba.obj ${CMAKE_CURRENT_BINARY_DIR}/calculadora.obj)
  add_executable(lc3_tests tests/test_isa.c)
  target_link_libraries(lc3_tests PRIVATE cpu_core assembler_lib)
  add_test(NAME isa COMMAND lc3_tests)
  set_tests_properties(isa PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(benchmark tests/benchmark.c)
  target_link_libraries(benchmark PRIVATE cpu_core)
  add_dependencies(benchmark benchmark_objects)
  add_test(NAME benchmark COMMAND benchmark ${CMAKE_CURRENT_BINARY_DIR}/funciones.obj ${CMAKE_CURRENT_BINARY_DIR}/benchmark.csv)
  add_executable(test_calculadora tests/test_calculadora.c)
  target_link_libraries(test_calculadora PRIVATE cpu_core assembler_lib)
  add_dependencies(test_calculadora benchmark_objects)
  add_test(NAME menu COMMAND test_calculadora ${CMAKE_CURRENT_BINARY_DIR}/calculadora_automatica.obj ${CMAKE_CURRENT_BINARY_DIR}/calculadora.obj)
  set_tests_properties(menu PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_exp_exhaustive tests/test_exp_exhaustive.c)
  target_link_libraries(test_exp_exhaustive PRIVATE cpu_core)
  if(NOT MSVC)
    target_link_libraries(test_exp_exhaustive PRIVATE m)
  endif()
  add_dependencies(test_exp_exhaustive benchmark_objects)
  add_test(NAME exp_65536 COMMAND test_exp_exhaustive ${CMAKE_CURRENT_BINARY_DIR}/funciones.obj)
  add_executable(test_exp_optimized tests/test_exp_optimized.c)
  target_link_libraries(test_exp_optimized PRIVATE cpu_core assembler_lib)
  if(NOT MSVC)
    target_link_libraries(test_exp_optimized PRIVATE m)
  endif()
  add_test(NAME exp_optimizada COMMAND test_exp_optimized tests/exp_previa/principal.asm tests/exp_actual.asm ${CMAKE_CURRENT_BINARY_DIR}/exp_comparacion.csv ${CMAKE_CURRENT_BINARY_DIR}/exp_optimizacion.json)
  set_tests_properties(exp_optimizada PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_shift_exhaustive tests/test_shift_exhaustive.c)
  target_link_libraries(test_shift_exhaustive PRIVATE cpu_core)
  add_test(NAME shift_3145728 COMMAND test_shift_exhaustive)
  add_executable(test_alu_extended tests/test_alu_extended.c)
  target_link_libraries(test_alu_extended PRIVATE cpu_core assembler_lib)
  add_test(NAME alu_integrada COMMAND test_alu_extended)
  add_executable(test_sqrt_equivalence tests/test_sqrt_equivalence.c)
  target_link_libraries(test_sqrt_equivalence PRIVATE cpu_core assembler_lib)
  add_dependencies(test_sqrt_equivalence benchmark_objects)
  add_test(NAME sqrt_equivalence COMMAND test_sqrt_equivalence ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/funciones.asm ${CMAKE_CURRENT_SOURCE_DIR}/benchmark/modulos ${CMAKE_CURRENT_BINARY_DIR}/sqrt_integrada.bin ${CMAKE_CURRENT_SOURCE_DIR}/tests/sqrt_reference.bin)
endif()
